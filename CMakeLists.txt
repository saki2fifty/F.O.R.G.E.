cmake_minimum_required(VERSION 3.24)
project(FORGE LANGUAGES C CXX)
set(FORGE_BUILD_ID "unassigned" CACHE STRING "Reserved UTC yymmdd-counter build identifier")
set(FORGE_SOURCE_COMMIT "unknown" CACHE STRING "Source commit packaged in this build")
if(NOT FORGE_BUILD_ID STREQUAL "unassigned" AND NOT FORGE_BUILD_ID MATCHES "^[0-9][0-9][0-9][0-9][0-9][0-9]-[0-9][0-9][0-9][0-9][0-9][0-9]+$")
 message(FATAL_ERROR "FORGE_BUILD_ID must be yymmdd-counter (at least six counter digits)")
endif()
if(NOT FORGE_SOURCE_COMMIT MATCHES "^([0-9a-f]+|unknown)$")
 message(FATAL_ERROR "Invalid source commit")
endif()
configure_file(cmake/build.hpp.in generated/forge/build.hpp @ONLY)
configure_file(cmake/build.json.in build.json @ONLY)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FORGE_BUILD_EDITOR "Build Windows D3D12 editor" ${WIN32})
include(cmake/dependencies.cmake)
add_library(forge_core src/schema.cpp src/services.cpp src/input.cpp src/project_paths.cpp src/project.cpp src/identity.cpp src/transform.cpp src/spatial_document.cpp src/scene_identity.cpp src/world.cpp src/builtins.cpp src/scene.cpp src/prefab.cpp src/module.cpp)
target_include_directories(forge_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(forge_core PUBLIC flecs_static nlohmann_json::nlohmann_json PRIVATE ${CMAKE_DL_LIBS})
if(WIN32)
 target_link_libraries(forge_core PRIVATE bcrypt)
endif()
if(MSVC)
 target_compile_options(forge_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(forge_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(forge_assets src/assets.cpp)
target_link_libraries(forge_assets PUBLIC forge_core)
# Authoring services are shared by editor and headless tools, never linked into runtime.
add_library(forge_authoring src/authoring.cpp src/prefab_authoring.cpp src/project_lease.cpp src/live_authoring.cpp)
if(WIN32)
 target_link_libraries(forge_authoring PRIVATE ws2_32 bcrypt)
endif()
target_link_libraries(forge_authoring PUBLIC forge_core forge_assets)
add_executable(forge_tools src/tools_main.cpp)
target_link_libraries(forge_tools PRIVATE forge_authoring)
add_library(forge_simulation src/runtime.cpp)
target_link_libraries(forge_simulation PUBLIC forge_core)
if(MSVC)
 target_compile_options(forge_simulation PRIVATE /W4 /permissive-)
else()
 target_compile_options(forge_simulation PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(forge_runtime src/runtime_main.cpp)
target_link_libraries(forge_runtime PRIVATE forge_simulation)
add_library(forge_sample MODULE samples/native/movement.c)
target_include_directories(forge_sample PRIVATE include)
set_target_properties(forge_sample PROPERTIES PREFIX "")
include(CTest)
if(FORGE_BUILD_EDITOR)
 if(NOT WIN32)
  message(FATAL_ERROR "The initial editor requires Windows/D3D12. Use -DFORGE_BUILD_EDITOR=OFF for portable core tests.")
 endif()
 include(cmake/editor.cmake)
endif()
if(BUILD_TESTING)
 add_executable(forge_services_tests tests/services_tests.cpp)
 target_link_libraries(forge_services_tests PRIVATE forge_simulation forge_assets)
 add_test(NAME core_services COMMAND forge_services_tests ${CMAKE_CURRENT_BINARY_DIR}/services-test-data)
 add_executable(forge_runtime_clock_tests tests/runtime_clock_tests.cpp)
 target_link_libraries(forge_runtime_clock_tests PRIVATE forge_simulation)
 add_test(NAME runtime_clock_and_interpolation COMMAND forge_runtime_clock_tests $<TARGET_FILE:forge_sample>)
 add_executable(forge_prefab_tests tests/prefab_tests.cpp)
 target_link_libraries(forge_prefab_tests PRIVATE forge_authoring)
 add_test(NAME structured_prefabs COMMAND forge_prefab_tests ${CMAKE_CURRENT_BINARY_DIR}/prefab-test-data)
 add_executable(forge_transform_tests tests/transform_tests.cpp)
 target_link_libraries(forge_transform_tests PRIVATE forge_authoring)
 add_test(NAME hierarchical_transforms COMMAND forge_transform_tests)
 add_executable(forge_identity_tests tests/identity_tests.cpp)
 target_link_libraries(forge_identity_tests PRIVATE forge_authoring)
 add_test(NAME persistent_identity COMMAND forge_identity_tests)
 add_executable(forge_world_tests tests/world_lifetime_tests.cpp)
 target_link_libraries(forge_world_tests PRIVATE forge_authoring)
 if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(forge_world_tests PRIVATE FORGE_WRAP_WORLD_INIT=1)
  target_link_options(forge_world_tests PRIVATE "LINKER:--wrap=ecs_init")
 endif()
 add_test(NAME world_lifetime COMMAND forge_world_tests)
 add_executable(forge_authoring_tests tests/authoring_api_tests.cpp)
 target_link_libraries(forge_authoring_tests PRIVATE forge_authoring)
 add_test(NAME authoring_api COMMAND forge_authoring_tests)
 add_executable(forge_tests tests/core_tests.cpp)
 target_link_libraries(forge_tests PRIVATE forge_core)
 add_test(NAME core COMMAND forge_tests $<TARGET_FILE:forge_sample>)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_executable(forge_live_tests tests/live_host.cpp)
 target_link_libraries(forge_live_tests PRIVATE forge_authoring)
 add_test(NAME authoring_live COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/live_test.py $<TARGET_FILE:forge_live_tests>)
 set_tests_properties(authoring_live PROPERTIES TIMEOUT 60)
 add_test(NAME authoring_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/authoring_cli_test.py $<TARGET_FILE:forge_tools>)
 add_test(NAME input_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/input_protocol_test.py $<TARGET_FILE:forge_runtime>)
 set_tests_properties(input_protocol PROPERTIES TIMEOUT 30)
 add_test(NAME runtime_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_test.py $<TARGET_FILE:forge_runtime> $<TARGET_FILE:forge_sample>)
 add_test(NAME native_iteration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native_test.py ${CMAKE_CURRENT_SOURCE_DIR} $<TARGET_FILE:forge_runtime> ${CMAKE_COMMAND} ${CMAKE_MAKE_PROGRAM})
 add_test(NAME plugin_packages COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/plugins_test.py ${CMAKE_CURRENT_SOURCE_DIR})
 set_tests_properties(core runtime_protocol native_iteration PROPERTIES TIMEOUT 90)
endif()
install(TARGETS forge_runtime RUNTIME DESTINATION bin COMPONENT Runtime)
install(FILES include/forge/module_api.h DESTINATION include/forge COMPONENT SDK)
install(FILES "${flecs_SOURCE_DIR}/LICENSE" DESTINATION share/forge/licenses RENAME flecs.txt COMPONENT Runtime)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION share/forge/licenses RENAME json.txt COMPONENT Runtime)

install(TARGETS forge_tools RUNTIME DESTINATION bin COMPONENT Tools)

option(FORGE_ENABLE_SANITIZERS "Enable address/undefined sanitizers for supported GNU/Clang core tooling" OFF)
if(FORGE_ENABLE_SANITIZERS)
 if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "FORGE_ENABLE_SANITIZERS requires GNU or Clang")
 endif()
 target_compile_options(forge_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(forge_core INTERFACE -fsanitize=address,undefined)
endif()

option(FORGE_DISABLE_PROFILING "Compile out core profiling scopes" OFF)
if(FORGE_DISABLE_PROFILING)
 target_compile_definitions(forge_core PUBLIC FORGE_DISABLE_PROFILING=1)
endif()

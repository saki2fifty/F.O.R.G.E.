cmake_minimum_required(VERSION 3.24)
project(ForgeReferenceGame LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ForgeNativeSdk CONFIG REQUIRED PATHS "${FORGE_NATIVE_SDK}/sdk" NO_DEFAULT_PATH)
add_library(reference_game MODULE gameplay.cpp)
target_link_libraries(reference_game PRIVATE ForgeNativeSdk::Client)
set_target_properties(reference_game PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE)
if(WIN32)
 forge_install_gameplay_runtime(reference_game NAME project.reference)
endif()

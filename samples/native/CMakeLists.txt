cmake_minimum_required(VERSION 3.24)
project(ForgeGameplay LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${FORGE_SDK}/include/forge/module_api.h")
 message(FATAL_ERROR "FORGE_SDK must contain include/forge/module_api.h")
endif()
add_library(gameplay MODULE gameplay.cpp)
target_include_directories(gameplay PRIVATE "${FORGE_SDK}/include")
set_target_properties(gameplay PROPERTIES PREFIX "")

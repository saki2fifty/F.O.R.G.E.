cmake_minimum_required(VERSION 3.24)
project(ForgeInternalSdkProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(ForgeNativeSdk CONFIG REQUIRED PATHS "${FORGE_NATIVE_SDK}/sdk" NO_DEFAULT_PATH)
add_library(gameplay MODULE probe.cpp)
target_link_libraries(gameplay PRIVATE ForgeNativeSdk::Client)
set_target_properties(gameplay PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE)
add_library(audio_gameplay MODULE audio_probe.cpp)
target_link_libraries(audio_gameplay PRIVATE ForgeNativeSdk::Client)
set_target_properties(audio_gameplay PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE)
add_library(animation_gameplay MODULE animation_probe.cpp)
target_link_libraries(animation_gameplay PRIVATE ForgeNativeSdk::Client)
set_target_properties(animation_gameplay PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE)
add_library(ui_gameplay MODULE ui_probe.cpp)
target_link_libraries(ui_gameplay PRIVATE ForgeNativeSdk::Client)
set_target_properties(ui_gameplay PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE)

add_library(combined_gameplay MODULE combined.cpp)
target_link_libraries(combined_gameplay PRIVATE ForgeNativeSdk::Client)
set_target_properties(combined_gameplay PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE)

# the source checkout retains one authoritative Gallery implementation.
set(FORGE_PREFAB_EXAMPLE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/SpawnPrefabs.cpp")
if(NOT EXISTS "${FORGE_PREFAB_EXAMPLE_SOURCE}")
 set(FORGE_PREFAB_EXAMPLE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../projects/FeatureGallery/CodeExamples/SpawnPrefabs.cpp")
endif()
add_library(prefab_gameplay MODULE prefab_spawn_probe.cpp "${FORGE_PREFAB_EXAMPLE_SOURCE}")
target_link_libraries(prefab_gameplay PRIVATE ForgeNativeSdk::Client)
set_target_properties(prefab_gameplay PROPERTIES PREFIX "" BUILD_RPATH_USE_ORIGIN TRUE)
set(FORGE_MATERIAL_EXAMPLE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT EXISTS "${FORGE_MATERIAL_EXAMPLE_INCLUDE}/ObjectMaterialValues.hpp")
 set(FORGE_MATERIAL_EXAMPLE_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../projects/FeatureGallery/CodeExamples")

add_library(material_animation_gameplay MODULE material_animation_probe.cpp)
